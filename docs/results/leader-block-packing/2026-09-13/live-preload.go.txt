const identity = "AzDzG3imFz5ihvetCcGixiwuieH7jfsaivnGDXgZo6Z8"
const genesis = "D5SkuirXeXeUTR7z2A65nzgU1GZ6uJ4vzf2tCy3Zt88b"
const rpcURL = "https://rpc.ag.validator1.net"

var root = flag.String("root", "/srv/mithril-leader-window-prefill-20260913/load", "private experiment directory")
var mode = flag.String("mode", "selfcheck", "init, selfcheck, simulate, or run")
var count = flag.Int("count", 32, "unique transactions; hard limit 200000")
var rate = flag.Int("rate", 100, "offered transactions/s; hard limit 150000")
var stage = flag.String("stage", "pilot", "unique result directory name")
var prefill = flag.Int("prefill-slots", 0, "send this many slots before leadership (0..20)")
var shape = flag.String("shape", "readonly-pair", "account-nonce, compute-budget or readonly-pair")
var transport = &http.Client{Timeout: 15 * time.Second}

func must(err error) {
	if err != nil {
		panic(err)
	}
}
func logEvent(v any) { b, e := json.Marshal(v); must(e); fmt.Println(string(b)) }
func save(path string, v any) {
	b, e := json.MarshalIndent(v, "", "  ")
	must(e)
	must(os.WriteFile(path, b, 0600))
}
func rpc(method string, params any, out any) error {
	b, e := json.Marshal(map[string]any{"jsonrpc": "2.0", "id": 1, "method": method, "params": params})
	if e != nil {
		return e
	}
	r, e := transport.Post(rpcURL, "application/json", bytes.NewReader(b))
	if e != nil {
		return e
	}
	defer r.Body.Close()
	var env struct {
		Result json.RawMessage `json:"result"`
		Error  json.RawMessage `json:"error"`
	}
	if e = json.NewDecoder(r.Body).Decode(&env); e != nil {
		return e
	}
	if len(env.Error) > 0 && string(env.Error) != "null" {
		return fmt.Errorf("RPC %s: %s", method, env.Error)
	}
	return json.Unmarshal(env.Result, out)
}
func keyPath(i int) string { return filepath.Join(*root, "keys", fmt.Sprintf("payer-%d.json", i)) }
func initKeys() {
	must(os.MkdirAll(filepath.Join(*root, "keys"), 0700))
	var pubs []string
	for i := 0; i < 8; i++ {
		_, p, e := ed25519.GenerateKey(rand.Reader)
		must(e)
		vals := make([]int, len(p))
		for j, b := range p {
			vals[j] = int(b)
		}
		b, e := json.Marshal(vals)
		must(e)
		f, e := os.OpenFile(keyPath(i), os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
		must(e)
		_, e = f.Write(b)
		must(e)
		must(f.Close())
		pubs = append(pubs, solana.PublicKeyFromBytes(p.Public().(ed25519.PublicKey)).String())
	}
	save(filepath.Join(*root, "payers.json"), pubs)
	logEvent(map[string]any{"payers": pubs})
}
func loadKeys() []ed25519.PrivateKey {
	keys := make([]ed25519.PrivateKey, 8)
	for i := range keys {
		b, e := os.ReadFile(keyPath(i))
		must(e)
		var a []int
		must(json.Unmarshal(b, &a))
		if len(a) != 64 {
			panic("invalid key length")
		}
		p := make([]byte, 64)
		for j, v := range a {
			if v < 0 || v > 255 {
				panic("invalid key byte")
			}
			p[j] = byte(v)
		}
		keys[i] = ed25519.PrivateKey(p)
	}
	return keys
}
func pairIndices(ordinal int) (int, int) {
	if ordinal < 0 || ordinal >= 128*127 {
		panic("pair ordinal outside pool capacity")
	}
	// A permutation distributes early transactions over the fixed account pool.
	n := (ordinal * 7919) % (128 * 127)
	a, b := n/127, n%127
	if b >= a {
		b++
	}
	return a, b
}
func loadPool() []solana.PublicKey {
	b, e := os.ReadFile(filepath.Join(*root, "pool.json"))
	must(e)
	var names []string
	must(json.Unmarshal(b, &names))
	if len(names) != 128 {
		panic("wrong pool length")
	}
	p := make([]solana.PublicKey, 128)
	seen := map[solana.PublicKey]bool{}
	for i, n := range names {
		p[i] = solana.MustPublicKeyFromBase58(n)
		if seen[p[i]] {
			panic("duplicate pool account")
		}
		seen[p[i]] = true
	}
	return p
}
func makePairWire(key ed25519.PrivateKey, hash solana.Hash, pool []solana.PublicKey, ordinal int) []byte {
	a, b := pairIndices(ordinal)
	payer := solana.PublicKeyFromBytes(key.Public().(ed25519.PublicKey))
	if payer == pool[a] || payer == pool[b] {
		panic("payer must be distinct from readonly pool")
	}
	msg := []byte{1, 0, 2, 3}
	msg = append(msg, payer[:]...)
	msg = append(msg, pool[a][:]...)
	msg = append(msg, pool[b][:]...)
	msg = append(msg, hash[:]...)
	msg = append(msg, 0)
	w := []byte{1}
	w = append(w, ed25519.Sign(key, msg)...)
	return append(w, msg...)
}

func makeWire(key ed25519.PrivateKey, hash solana.Hash, nonce [32]byte) []byte {
	// One writable signer, one unique readonly unsigned account, no instructions.
	msg := []byte{1, 0, 1, 2}
	msg = append(msg, key.Public().(ed25519.PublicKey)...)
	msg = append(msg, nonce[:]...)
	msg = append(msg, hash[:]...)
	msg = append(msg, 0)
	sig := ed25519.Sign(key, msg)
	w := []byte{1}
	w = append(w, sig...)
	return append(w, msg...)
}
func makeBudgetWire(key ed25519.PrivateKey, hash solana.Hash, limit uint32) []byte {
	program := solana.MustPublicKeyFromBase58("ComputeBudget111111111111111111111111111111")
	msg := []byte{1, 0, 1, 2}
	msg = append(msg, key.Public().(ed25519.PublicKey)...)
	msg = append(msg, program[:]...)
	msg = append(msg, hash[:]...)
	msg = append(msg, 1, 1, 0, 5, 2) // one instruction: program 1, no accounts, 5 data bytes, SetComputeUnitLimit
	msg = binary.LittleEndian.AppendUint32(msg, limit)
	w := []byte{1}
	w = append(w, ed25519.Sign(key, msg)...)
	return append(w, msg...)
}
func check(w []byte) {
	t, e := solana.TransactionFromBytes(w)
	must(e)
	if (len(w) != 166 && len(w) != 174 && len(w) != 198) || len(t.Signatures) != 1 || len(t.Message.Instructions) > 1 {
		panic("wrong transaction shape")
	}
	if !ed25519.Verify(ed25519.PublicKey(t.Message.AccountKeys[0][:]), w[65:], w[1:65]) {
		panic("invalid signature")
	}
}
func getSlot() uint64 {
	var s uint64
	must(rpc("getSlot", []any{map[string]any{"commitment": "processed"}}, &s))
	return s
}
func getHash() (solana.Hash, uint64) {
	var r struct {
		Value struct {
			Blockhash            string
			LastValidBlockHeight uint64
		}
	}
	must(rpc("getLatestBlockhash", []any{map[string]any{"commitment": "confirmed"}}, &r))
	h, e := solana.HashFromBase58(r.Value.Blockhash)
	must(e)
	return h, r.Value.LastValidBlockHeight
}
func selectWindow() uint64 {
	var e struct{ AbsoluteSlot, SlotIndex uint64 }
	must(rpc("getEpochInfo", []any{}, &e))
	var sched map[string][]uint64
	must(rpc("getLeaderSchedule", []any{nil, map[string]any{"identity": identity}}, &sched))
	base := e.AbsoluteSlot - e.SlotIndex
	for _, s := range sched[identity] {
		abs := base + s
		if abs%4 == 0 && abs > e.AbsoluteSlot+65 {
			return abs
		}
	}
	panic("no suitable leader window in this epoch")
}
func awaitSlot(target uint64) uint64 {
	deadline := time.Now().Add(10 * time.Minute)
	for time.Now().Before(deadline) {
		s := getSlot()
		if s >= target {
			return s
		}
		if target-s > 30 {
			time.Sleep(2 * time.Second)
		} else {
			time.Sleep(75 * time.Millisecond)
		}
	}
	panic("wait deadline")
}
func makePhasedPairWire(keys []ed25519.PrivateKey, first, second solana.Hash, pool []solana.PublicKey, i int) []byte {
	if i < 0 || i >= 200000 || first == second {
		panic("invalid phased load input")
	}
	txHash, ordinal := first, i/8
	if i >= 120000 {
		txHash, ordinal = second, (i-120000)/8
	}
	return makePairWire(keys[i%8], txHash, pool, ordinal)
}

func main() {
	flag.Parse()
	if *shape != "readonly-pair" && *shape != "account-nonce" && *shape != "compute-budget" {
		panic("invalid shape")
	}
	if *mode == "init" {
		initKeys()
		return
	}
	if *count < 1 || *count > 200000 || *rate < 1 || *rate > 150000 || *prefill < 0 || *prefill > 20 {
		panic("outside experiment bounds")
	}
	if filepath.Base(*stage) != *stage {
		panic("invalid stage")
	}
	if *mode == "selfcheck" {
		_, p, e := ed25519.GenerateKey(rand.Reader)
		must(e)
		w := makeWire(p, solana.Hash{}, sha256.Sum256([]byte("test")))
		check(w)
		check(makeBudgetWire(p, solana.Hash{}, 1000))
		pool := make([]solana.PublicKey, 128)
		for i := range pool {
			pool[i] = solana.PublicKey{byte(i + 1), 77}
		}
		seen := map[[64]byte]bool{}
		for i := 0; i < 128*127; i++ {
			a, b := pairIndices(i)
			var pair [64]byte
			copy(pair[:32], pool[a][:])
			copy(pair[32:], pool[b][:])
			if a == b || seen[pair] {
				panic("pair collision")
			}
			seen[pair] = true
		}
		for _, i := range []int{0, 1, 126, 127, 14999, 16255} {
			check(makePairWire(p, solana.Hash{}, pool, i))
		}
		logEvent(map[string]any{"wire_bytes": 198, "signatures": 1, "signature_verified": true, "unique_pairs": len(seen)})
		return
	}
	var g string
	must(rpc("getGenesisHash", []any{}, &g))
	if g != genesis {
		panic("unexpected cluster genesis")
	}
	keys := loadKeys()
	pool := loadPool()
	if *mode == "simulate" {
		h, _ := getHash()
		w := makeWire(keys[0], h, sha256.Sum256([]byte("simulation")))
		if *shape == "compute-budget" {
			w = makeBudgetWire(keys[0], h, 1000)
		}
		if *shape == "readonly-pair" {
			w = makePairWire(keys[0], h, pool, 14999)
		}
		check(w)
		var sim any
		must(rpc("simulateTransaction", []any{base64.StdEncoding.EncodeToString(w), map[string]any{"encoding": "base64", "sigVerify": true, "commitment": "confirmed"}}, &sim))
		save(filepath.Join(*root, *shape+"-simulation.json"), sim)
		logEvent(sim)
		return
	}
	if *mode != "run" {
		panic("unknown mode")
	}
	// Claim the stage before signing/sending. No automatic replay of uncertain runs.
	out := filepath.Join(*root, *stage)
	must(os.Mkdir(out, 0700))
	// Bound all experiment sends by a durable count, including uncertain prior runs.
	ledger := filepath.Join(*root, "authorized-count.json")
	used := 0
	if b, e := os.ReadFile(ledger); e == nil {
		must(json.Unmarshal(b, &used))
	} else if !os.IsNotExist(e) {
		must(e)
	}
	if used+*count > 200000 {
		panic("cumulative maximum fee 1.0 testnet SOL exceeded")
	}
	save(ledger, used+*count)
	window := selectWindow()
	logEvent(map[string]any{"event": "waiting_to_prepare", "leader_window": window, "now_slot": getSlot(), "count": *count, "offered_tps": *rate})
	awaitSlot(window - 55)
	hash, lastHeight := getHash()
	hashLedger := filepath.Join(*root, "used-blockhashes.json")
	usedHashes := map[string]bool{}
	if data, err := os.ReadFile(hashLedger); err == nil {
		must(json.Unmarshal(data, &usedHashes))
	} else if !os.IsNotExist(err) {
		must(err)
	}
	if usedHashes[hash.String()] {
		panic("blockhash already used by this experiment")
	}
	usedHashes[hash.String()] = true
	save(hashLedger, usedHashes)
	secondHash, secondHeight := hash, lastHeight
	if *count > 120000 {
		for attempts := 0; attempts < 40; attempts++ {
			time.Sleep(100 * time.Millisecond)
			secondHash, secondHeight = getHash()
			if secondHash != hash {
				break
			}
		}
		if secondHash == hash || usedHashes[secondHash.String()] {
			panic("no fresh top-up blockhash")
		}
		usedHashes[secondHash.String()] = true
		save(hashLedger, usedHashes)
	}
	nonce := make([]byte, 32)
	_, e := rand.Read(nonce)
	must(e)
	wires := make([][]byte, *count)
	next := atomic.Int64{}
	var wg sync.WaitGroup
	beg := time.Now()
	for k := 0; k < 4; k++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for {
				i := int(next.Add(1) - 1)
				if i >= len(wires) {
					return
				}
				n := append([]byte(nil), nonce...)
				n = binary.LittleEndian.AppendUint64(n, uint64(i))
				wires[i] = makeWire(keys[i%8], hash, sha256.Sum256(n))
				if *shape == "compute-budget" {
					wires[i] = makeBudgetWire(keys[i%8], hash, uint32(1000+i))
				}
				if *shape == "readonly-pair" {
					if len(wires) > 120000 {
						wires[i] = makePhasedPairWire(keys, hash, secondHash, pool, i)
					} else {
						wires[i] = makePairWire(keys[i%8], hash, pool, i/8)
					}
				}
			}
		}()
	}
	wg.Wait()
	check(wires[0])
	check(wires[len(wires)-1])
	if len(wires) > 120000 {
		check(wires[120000])
	}
	ids := make([]string, len(wires))
	for i, w := range wires {
		ids[i] = solana.SignatureFromBytes(w[1:65]).String()
	}
	save(filepath.Join(out, "signatures.json"), ids)
	meta := map[string]any{"window": window, "count": *count, "offered_tps": *rate, "transaction_bytes": len(wires[0]), "shape": *shape, "signatures_per_transaction": 1, "prefill_slots": *prefill, "blockhash": hash.String(), "last_valid_block_height": max(lastHeight, secondHeight), "topup_blockhash": secondHash.String(), "prefill_count": *count, "topup_count": 0, "topup_trigger": "none; entire workload preloaded", "prepare_seconds": time.Since(beg).Seconds(), "sender": "127.0.0.1:8011; GOMAXPROCS=2; signatures prepared before send"}
	save(filepath.Join(out, "metadata.json"), meta)
	if getSlot() >= window-uint64(*prefill) {
		panic("missed preparation deadline; no sends")
	}
	client, e := testutils.NewClient()
	must(e)
	// Four connections and four streams each respect the validator's existing limits.
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()
	type senderFn func(context.Context, []byte) error
	senders := make([]senderFn, 4)
	var closers []func()
	for i := range senders {
		c, e := client.Dial(ctx, "127.0.0.1:8011")
		must(e)
		senders[i] = func(cctx context.Context, w []byte) error { return client.Send(cctx, c, w) }
		closers = append(closers, func() { _ = c.CloseWithError(0, "") })
	}
	defer func() {
		for _, f := range closers {
			f()
		}
	}()
	observed := awaitSlot(window - uint64(*prefill))
	if observed > window+1 {
		panic("missed send window; no sends")
	}
	var height uint64
	must(rpc("getBlockHeight", []any{map[string]any{"commitment": "confirmed"}}, &height))
	if height+10 >= lastHeight {
		panic("blockhash too old")
	}
	logEvent(map[string]any{"event": "sending", "observed_slot": observed, "leader_window": window})
	start := time.Now()
	jobs := make(chan int, 512)
	var sent, errors, skipped atomic.Uint64
	var firstErr string
	var errMu sync.Mutex
	var endMu sync.Mutex
	lastSend := start
	sendCtx, stop := context.WithTimeout(context.Background(), 3*time.Second)
	defer stop()
	for k := 0; k < 16; k++ {
		wg.Add(1)
		go func(worker int) {
			defer wg.Done()
			for i := range jobs {
				if sendCtx.Err() != nil {
					skipped.Add(1)
					continue
				}
				e := senders[worker%4](sendCtx, wires[i])
				if e != nil {
					errors.Add(1)
					errMu.Lock()
					if firstErr == "" {
						firstErr = e.Error()
					}
					errMu.Unlock()
				} else {
					sent.Add(1)
					endMu.Lock()
					lastSend = time.Now()
					endMu.Unlock()
				}
			}
		}(k)
	}
	for i := 0; i < len(wires); {

		if sendCtx.Err() != nil {
			skipped.Add(uint64(len(wires) - i))
			break
		}
		end := i + 150
		if end > len(wires) {
			end = len(wires)
		}
		target := start.Add(time.Duration(float64(i) / float64(*rate) * float64(time.Second)))
		if d := time.Until(target); d > 0 {
			time.Sleep(d)
		}
		for ; i < end; i++ {
			select {
			case jobs <- i:
			case <-sendCtx.Done():
				skipped.Add(1)
			}
		}
	}
	close(jobs)
	wg.Wait()
	summary := map[string]any{"sent": sent.Load(), "send_errors": errors.Load(), "not_sent": skipped.Load(), "first_error": firstErr, "start_utc": start.UTC().Format(time.RFC3339Nano), "end_utc": lastSend.UTC().Format(time.RFC3339Nano), "send_seconds": lastSend.Sub(start).Seconds(), "end_slot": getSlot(), "target_window": window}
	save(filepath.Join(out, "send.json"), summary)
	logEvent(summary)
	// Let already-written stream data be acknowledged before closing QUIC connections.
	time.Sleep(500 * time.Millisecond)
}
